# Deep Teal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Dots gets the Deep Teal look: a light and a dark theme with a System/Light/Dark switch that also themes the toasts and the title bar, a calmer document-style conversation (bubbles, a meta line, tool steps as chips, soft failures), and consistent polish (focus rings, one page header, empty states), without moving any screen's layout.
**Architecture:** The colours are shadcn's CSS variables in `ui/src/index.css` (`:root` light, `.dark` dark) plus new semantic ones (`--needs*`, `--bubble*`, `--success`) exposed to Tailwind through `@theme inline`; `ui/src/lib/tokens.ts` mirrors them so one test holds the two together and measures every pair for WCAG AA. `ui/src/lib/theme.ts` keeps the choice in `localStorage` (`dots.theme`), applies it to `<html>` and the Tauri title bar, follows Windows while System is chosen, and feeds the toaster; `ui/public/theme-boot.js` applies the same rule before the first paint. The rest is small class edits in the components the spec names.
**Tech Stack:** React 19.3, Tailwind CSS 4.3 (`@tailwindcss/vite`), shadcn components with `cn` 0.4 (tailwind-merge compatible), lucide-react 1.55, sonner 2.0.8, @tauri-apps/api 2.12.1 (`getCurrentWindow().setTheme`), Tauri 2 capabilities; Vitest 5.0 with Testing Library, jsdom and the fake backend (`ui/src/test/backend.ts`); Bun 1.4.3. No new dependency.
**Spec:** docs/superpowers/specs/2026-10-22-deep-teal-design.md (sections 1 to 7; section 8 is out of scope; Task 7 fills section 9)

The inputs and failure modes the spec implies that are most likely to bite a person, each pinned to the test that covers it:

1. **Storage that cannot be read or holds junk** (a locked-down profile, an old or hand-edited `dots.theme`). import { readFileSync } from "node:fs";
```ts
import path from "node:path";

import { describe, expect, it } from "vitest";

import { CONTRAST_PAIRS, THEMES, contrastRatio, pairColours, type ThemeName } from "@/lib/tokens";

const css = readFileSync(path.resolve(import.meta.dirname, "../index.css"), "utf8");

/// The declarations of the block that starts with `selector {` at the start of a line.
function block(selector: string): Map<string, string> {
  const start = css.search(new RegExp(`^${selector.replace(".", "\.")} \{`, "m"));
  expect(start, `${selector} block`).toBeGreaterThanOrEqual(0);
  const body = css.slice(start, css.indexOf("}", start));
  return new Map(
    [...body.matchAll(/--([\w-]+):\s*([^;]+);/g)].map((m) => [m[1]!, m[2]!.trim()]),
  );
}

const SELECTOR: Record<ThemeName, string> = { light: ":root", dark: ".dark" };

describe("the Deep Teal tokens", () => {
  it("index.css declares every token with the value of the table, in both themes", () => {
    for (const theme of ["light", "dark"] as const) {
      const declared = block(SELECTOR[theme]);
      for (const [name, value] of Object.entries(THEMES[theme])) {
        expect(declared.get(name), `${theme} --${name}`).toBe(value);
      }
    }
  });

  it("measures contrast as WCAG does", () => {
    expect(contrastRatio("#000000", "#ffffff")).toBeCloseTo(21, 5);
    expect(contrastRatio("#ffffff", "#ffffff")).toBeCloseTo(1, 5);
    expect(contrastRatio("#767676", "#ffffff")).toBeCloseTo(4.54, 2);
  });

  it("gives every text and focus ring its AA contrast, in both themes", () => {
    const failures: string[] = [];
    for (const theme of ["light", "dark"] as const) {
      for (const pair of CONTRAST_PAIRS) {
        if (pair.theme !== undefined && pair.theme !== theme) continue;
        const [text, background] = pairColours(THEMES[theme], pair);
        const ratio = contrastRatio(text, background);
        if (ratio < pair.min) failures.push(`${theme}: ${pair.label} is ${ratio.toFixed(2)}`);
      }
    }
    expect(failures).toEqual([]);
  });
});
```

```ts
function rgb(hex: string): [number, number, number] {
  const m = /^#([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})$/i.exec(hex);
  if (!m) throw new Error(`not a #rrggbb colour: ${hex}`);
  return [parseInt(m[1]!, 16), parseInt(m[2]!, 16), parseInt(m[3]!, 16)];
}

const toHex = (channels: number[]) =>
  `#${channels.map((c) => Math.round(c).toString(16).padStart(2, "0")).join("")}`;

/// WCAG 2.2 relative luminance of a `#rrggbb` colour.
function luminance(hex: string): number {
  const [r, g, b] = rgb(hex).map((c) => {
    const s = c / 255;
    return s <= 0.04045 ? s / 12.92 : ((s + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * r! + 0.7152 * g! + 0.0722 * b!;
}

/// WCAG 2.2 contrast ratio of two `#rrggbb` colours, 1 to 21.
export function contrastRatio(a: string, b: string): number {
  const [hi, lo] = [luminance(a), luminance(b)].sort((x, y) => y - x);
  return (hi! + 0.05) / (lo! + 0.05);
}

/// `colour` laid at `alpha` over `over`, as the browser paints it.
export function composite(colour: string, alpha: number, over: string): string {
  const top = rgb(colour);
  const bottom = rgb(over);
  return toHex(top.map((c, i) => c * alpha + bottom[i]! * (1 - alpha)));
}

/// The text colour and the background colour of `pair` in `theme`.
export function pairColours(
  theme: Record<TokenName, string>,
  pair: ContrastPair,
): [string, string] {
  const background =
    typeof pair.on === "string"
      ? theme[pair.on]
      : composite(theme[pair.on.tint], pair.on.alpha, theme[pair.on.over]);
  return [theme[pair.text], background];
}
```

```ts
import { getCurrentWindow } from "@tauri-apps/api/window";
import { useSyncExternalStore } from "react";

/// The person's choice of theme (deep-teal spec section 3), kept per device in localStorage.
export type ThemeChoice = "system" | "light" | "dark";
export type Theme = "light" | "dark";

export const THEME_KEY = "dots.theme";
export const THEME_CHOICES: readonly ThemeChoice[] = ["system", "light", "dark"];
const DARK_QUERY = "(prefers-color-scheme: dark)";

/// The choice made in this session, for when storage cannot be read.
let sessionChoice: ThemeChoice = "system";
/// What re-renders when the theme on screen changes (the toaster).
const listeners = new Set<() => void>();

const parseChoice = (stored: string | null): ThemeChoice =>
  THEME_CHOICES.find((choice) => choice === stored) ?? "system";

/// The choice kept on this device: System when there is none, or it is not one of the three.
/// Storage that cannot be read counts as System, or as what was chosen in this session.
export function themeChoice(): ThemeChoice {
  try {
    return parseChoice(window.localStorage.getItem(THEME_KEY));
  } catch {
    return sessionChoice;
  }
}

/// The theme a choice shows. `public/theme-boot.js` applies the same rule before the first
/// paint.
export function resolvedTheme(choice: ThemeChoice, prefersDark: boolean): Theme {
  return choice === "system" ? (prefersDark ? "dark" : "light") : choice;
}

function prefersDark(): boolean {
  return typeof window.matchMedia === "function" && window.matchMedia(DARK_QUERY).matches;
}

/// Shows `choice` (by default the one kept): the `dark` class and the colour scheme on <html>,
/// the window's title bar (System lets Windows choose), and whatever follows the theme.
export function applyTheme(choice: ThemeChoice = themeChoice()): Theme {
  const theme = resolvedTheme(choice, prefersDark());
  const root = document.documentElement;
  root.classList.toggle("dark", theme === "dark");
  root.style.colorScheme = theme;
  try {
    getCurrentWindow()
      .setTheme(choice === "system" ? null : choice)
      .catch(() => {});
  } catch {
    // No Tauri window (a test, a browser): the page is themed anyway.
  }
  for (const listener of listeners) listener();
  return theme;
}

/// Keeps `choice` on this device and shows it at once. A choice that cannot be kept still
/// holds for this session.
export function setThemeChoice(choice: ThemeChoice): void {
  sessionChoice = choice;
  try {
    window.localStorage.setItem(THEME_KEY, choice);
  } catch {
    // Kept for this session only.
  }
  applyTheme(choice);
}

/// Shows the kept choice and, while it is System, follows Windows when its theme changes.
/// Returns what stops following.
export function startTheme(): () => void {
  applyTheme();
  if (typeof window.matchMedia !== "function") return () => {};
  const query = window.matchMedia(DARK_QUERY);
  const follow = () => {
    if (themeChoice() === "system") applyTheme("system");
  };
  query.addEventListener("change", follow);
  return () => query.removeEventListener("change", follow);
}

function subscribe(listener: () => void) {
  listeners.add(listener);
  return () => {
    listeners.delete(listener);
  };
}

const shownTheme = (): Theme =>
  document.documentElement.classList.contains("dark") ? "dark" : "light";

/// The theme on screen, for what React draws (the toasts).
export function useResolvedTheme(): Theme {
  return useSyncExternalStore(subscribe, shownTheme);
}
```

```js
// this file as a classic script in its <head>, so it runs before the page is drawn. It is a
// file, not an inline script, because the app's content security policy only runs scripts of
// the app itself. `src/lib/theme.ts` holds the same rule; `src/lib/theme.test.tsx` runs this
// file against it.
function dotsBootTheme(storage, prefersDark, root) {
  var choice = null;
  try {
    choice = storage.getItem("dots.theme");
  } catch (e) {
    // Storage that cannot be read counts as System.
  }
  var dark = choice === "dark" || (choice !== "light" && prefersDark);
  root.classList.toggle("dark", dark);
  root.style.colorScheme = dark ? "dark" : "light";
}

    // Blocked storage: System.

    typeof window.matchMedia === "function" &&
    window.matchMedia("(prefers-color-scheme: dark)").matches;
  dotsBootTheme(storage, prefersDark, document.documentElement);
```

```tsx
/// technology.
const GLYPH: Record<string, string> = {
  done: "✓",
  failed: "✕",
  running: "…",
  "no result": "–",

/// One tool step as a small rounded chip (its tool, then a mark) that opens to the call and its
/// result. A call sent for approval has its chip in the needs colour.
function ToolChip(props: { item: ToolItem; active: boolean; owners: ReadonlyMap<string, string> }) {
  const { item, active, owners } = props;
  const said = outcome(item.result, active);
  const tone = parked(item.result) ? "needs" : item.result?.isError ? "failed" : "plain";
  const mark =
    tone === "failed"
      ? "text-destructive"
      : tone === "needs"
        ? undefined
        : said === "done"
          ? "text-success"
          : "text-muted-foreground";
  return (
    <details data-tone={tone} className="open:basis-full">
      <summary
        className={cn(
          "inline-flex cursor-pointer list-none items-center gap-1.5 rounded-full border px-2.5 py-0.5 text-xs [&::-webkit-details-marker]:hidden",
          tone === "needs" ? "border-needs-border bg-needs-surface text-needs" : "bg-muted",
        )}
      >
        <span className="font-medium" title={item.tool}>
          {mcpToolLabel(item.tool, owners)}
        </span>
        <span aria-hidden className={mark}>
          {GLYPH[said]}
        </span>
        <span className="sr-only">{said}</span>
      </summary>
      <div className="mt-1.5 flex flex-col gap-1 rounded-lg border p-2">
        <p className="truncate font-mono text-xs text-muted-foreground">{toolPreview(item.input)}</p>
        <pre className="overflow-x-auto text-xs whitespace-pre-wrap">
          {JSON.stringify(item.input, null, 2)}
        </pre>
        {item.result && (
          <pre
            className={`max-h-80 overflow-auto border-t pt-1 text-xs whitespace-pre-wrap ${outcomeColour(
              item.result,
            )}`}
          >
            {item.result.output}
          </pre>
        )}
      </div>
    </details>
  );
}

/// What a run's section shows, in order: consecutive tool steps as one row of chips, every
/// other item by itself.
type Block =
  | { kind: "tools"; seq: number; items: ToolItem[] }
  | { kind: "item"; item: TranscriptItem };

function blocks(items: TranscriptItem[]): Block[] {
  const out: Block[] = [];
  for (const item of items) {
    const last = out.at(-1);
    if (item.type === "tool" && last?.kind === "tools") last.items.push(item);
    else if (item.type === "tool") out.push({ kind: "tools", seq: item.seq, items: [item] });
    else out.push({ kind: "item", item });
  }
  return out;
}
```
